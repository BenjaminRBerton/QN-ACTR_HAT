;;;;;;;;;;; PARAMETERS

(sgp	
	:visual-attention-latency	0.085	;This parameter specifies how long a visual attention shift will take in seconds.  The default value is .085.
	:imaginal-delay			0.2	;a non-negative number	The imaginal-delay parameter controls how long it takes a request or modification request to the imaginal buffer to complete.  It can be set to a non-negative time (in seconds) and defaults to .2.
)





;;;;;;;;;;; CHUNKS

(chunk-type  checklist-procedure			
	procedure-name
	step			
	phase

)

(chunk-type  checklist-item-representation
	item-name
)

(chunk-type  aircraft-component-status-representation
	aircraft-component-name
	status				
)



(chunk-type  pilot-control		;; control pitch, yaw, roll while flying	
	perceived-pitch-old
	perceived-yaw-old
	perceived-roll-old
	time-old
	pitch-controller-value
	yaw-controller-value
	roll-controller-value
	step			
)




(add-dm	(pilot-goal 
		isa 		pilot-control
		perceived-pitch-old	0
		perceived-yaw-old	0
		perceived-roll-old	0
		time-old		0
		pitch-controller-value	0
		yaw-controller-value	0
		roll-controller-value	0
		step			1
		)
)

(goal-focus pilot-goal)




;;;;;;;;;;; PRODUCTION RULES

(p pilot-control-attend-outside		;; visual flight rules VFR, see outside, control pitch, yaw, roll
   =goal>
      isa 	pilot-control
      step 	1
==>
 +visual-location>
  isa		visual-location
  screen-x	20			; representing outside
  screen-y	20

   =goal>
      step 2	
)

(p pilot-control-perceive-attitude
   =goal>
      isa 	pilot-control
      step 	2
   =visual-location>
!bind! =time (get-clock-time)     ;; the short-cut way  
!bind! =p (x-plane-getdref  sim/cockpit2/gauges/indicators/pitch_electric_deg_pilot)
!bind! =y (x-plane-getdref  sim/cockpit2/gauges/indicators/heading_vacuum_deg_mag_pilot)	;heading is yaw
!bind! =r (x-plane-getdref  sim/cockpit2/gauges/indicators/roll_electric_deg_pilot)
==>	
   =goal>
	perceived-pitch-old	=p
	perceived-yaw-old	=y
	perceived-roll-old	=r
	time-old		=time	
	step	3
   +visual-location>
      isa  visual-location-world3d-piloting
      kind outside
)


(p pilot-control-action
   =goal>
      	isa 	pilot-control
      	step 	3
	perceived-pitch-old	=p-old
	perceived-yaw-old	=y-old
	perceived-roll-old	=r-old
	time-old		=time-old
	pitch-controller-value	=p-control-old
	yaw-controller-value	=y-control-old
	roll-controller-value	=r-control-old		
 +visual-location>
  isa		visual-location
  screen-x	20			; representing outside
  screen-y	20

   ?manual>
      state   free
!bind! =time (get-clock-time)     ;; the short-cut way  
!bind! =p (x-plane-getdref  sim/cockpit2/gauges/indicators/pitch_electric_deg_pilot)
!bind! =y (x-plane-getdref  sim/cockpit2/gauges/indicators/heading_vacuum_deg_mag_pilot)	;heading is yaw
!bind! =r (x-plane-getdref  sim/cockpit2/gauges/indicators/roll_electric_deg_pilot)

==>

!bind! =p-control  (x-plane-do-pitch-control =p-old  =p   =time-old  =time	=p-control-old	0.0)		; the last parameter is a dummy, place holder.
!bind! =y-control  (x-plane-do-yaw-control   =y-old  =y   =time-old  =time	=y-control-old	0.0)
!bind! =r-control  (x-plane-do-roll-control  =r-old  =r   =time-old  =time	=r-control-old	0.0)
      
   =goal>
      	isa 	pilot-control
      	step 	1
	pitch-controller-value	=p-control
	yaw-controller-value	=y-control
	roll-controller-value	=r-control	
    +manual>
     isa 			customized-manual-action		
     name			x-plane-button-action
     preparation-duration	0.050
     initiation-duration	0.050
     execution-duration	    	0.050                 
     finish-duration		0.0
	;; this manual action is just for making the manual module busy for the duration.

)
